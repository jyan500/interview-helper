# Production migration — durable storage on Supabase Postgres (Phase A).
#   sqlalchemy[asyncio] : async ORM/core; the [asyncio] extra pulls in the async engine bits
#   asyncpg             : the async Postgres driver SQLAlchemy talks through (postgresql+asyncpg://)
#   alembic             : schema migrations (versioned DDL), so the DB shape is reproducible
sqlalchemy[asyncio]>=2.0,<3

# Auth — verify Supabase-issued JWTs on the backend (Phase B) via the project's JWKS endpoint.
# THE [crypto] EXTRA IS NOT OPTIONAL: bare PyJWT is pure-python and can only do HMAC (HS256).
# Supabase signs ours with ES256 (elliptic curve), which needs the `cryptography` package, and
# without it every verification dies on MissingCryptographyError. It happens to be in the venv
# already as a transitive dep of Authlib/google-auth — i.e. by luck, not by declaration, which
# is exactly the kind of thing that works locally and breaks on a clean Phase G deploy.
pyjwt[crypto]>=2.8
